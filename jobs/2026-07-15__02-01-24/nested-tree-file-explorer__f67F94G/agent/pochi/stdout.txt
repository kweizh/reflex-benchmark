import reflex as rx
import pydantic
from typing import List

class TreeNode(pydantic.BaseModel):
    name: str
    is_dir: bool
    children: List['TreeNode'] = []
    path: str = ''

TreeNode.model_rebuild()

def build_tree():
    return TreeNode(
        name='root',
        is_dir=True,
        path='root',
        children=[
            TreeNode(name='README.md', is_dir=False, path='root/README.md'),
        ],
    )

class State(rx.State):
    expanded: List[str] = ['root']
    selected: str = ''
    root_node: TreeNode = pydantic.Field(default_factory=build_tree)
    
    @rx.var
    def total_files(self) -> int:
        def count(node):
            if not node.is_dir:
                return 1
            return sum(count(c) for c in node.children)
        return count(self.root_node)

@rx.memo
def node_view(node: rx.Var[TreeNode], expanded: rx.Var[List[str]], depth: rx.Var[int] = 0) -> rx.Component:
    return rx.box(
        rx.text('  ' * depth),
        rx.cond(
            node.is_dir,
            rx.button(node.name, on_click=lambda: State.toggle_expand(node.path)),
            rx.button(node.name, on_click=lambda: State.select_file(node.path)),
        ),
        rx.cond(
            node.is_dir & node.path.contain(expanded),
            rx.foreach(
                node.children,
                lambda child: node_view(child, expanded, depth + 1),
            ),
        ),
    )

def index() -> rx.Component:
    return rx.container(
        rx.vstack(
            rx.text(f'Total files: {State.total_files}'),
            node_view(State.root_node, State.expanded, 0),
        ),
    )

def toggle_expand(self, node_id: str):
    if node_id in self.expanded:
        self.expanded = [x for x in self.expanded if x != node_id]
    else:
        self.expanded = self.expanded + [node_id]

def select_file(self, node_id: str):
    self.selected = node_id

State.toggle_expand = toggle_expand
State.select_file = select_file

app = rx.App()
app.add_page(index)
print('App ok')
