import reflex as rx
from typing import List

class State(rx.State):
    content: str = ''
    _undo_stack: List[str] = []
    _redo_stack: List[str] = []

    @rx.var
    def can_undo(self) -> bool:
        return len(self._undo_stack) > 0

    @rx.var
    def can_redo(self) -> bool:
        return len(self._redo_stack) > 0

    @rx.var
    def history_label(self) -> str:
        return f'Undo depth: {len(self._undo_stack)} | Redo depth: {len(self._redo_stack)}'

    def commit(self, value: str):
        # Push current to undo, set new content, clear redo
        self._undo_stack.append(self.content)
        # Trim to 50
        if len(self._undo_stack) > 50:
            self._undo_stack = self._undo_stack[-50:]
        self.content = value
        self._redo_stack = []

    def undo(self):
        if not self._undo_stack:
            return
        prev = self._undo_stack.pop()
        self._redo_stack.append(self.content)
        self.content = prev

    def redo(self):
        if not self._redo_stack:
            return
        prev = self._redo_stack.pop()
        self._undo_stack.append(self.content)
        self.content = prev

# Check
print('Has commit:', hasattr(State, 'commit'))
print('Has undo:', hasattr(State, 'undo'))
print('Has redo:', hasattr(State, 'redo'))

    @property
    def can_redo(self) -> bool:
        return len(self._redo_stack) > 0

    @property
    def history_label(self) -> str:
        return (
            f'Undo depth: {len(self._undo_stack)} | '
            f'Redo depth: {len(self._redo_stack)}'
        )

    def set_input_value(self, value):
        self.input_value = value

    def commit(self):
        self._undo_stack.append(self.content)
        if len(self._undo_stack) > 50:
            self._undo_stack = self._undo_stack[-50:]
        self.content = self.input_value
        self._redo_stack = []

    def undo(self):
        if not self._undo_stack:
            return
        previous = self._undo_stack.pop()
        self._redo_stack.append(self.content)
        self.content = previous
        self.input_value = previous

    def redo(self):
        if not self._redo_stack:
            return
        value = self._redo_stack.pop()
        self._undo_stack.append(self.content)
        if len(self._undo_stack) > 50:
            self._undo_stack = self._undo_stack[-50:]
        self.content = value
        self.input_value = value
