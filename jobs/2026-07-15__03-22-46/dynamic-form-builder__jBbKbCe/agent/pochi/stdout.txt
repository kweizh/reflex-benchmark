A user composes a form by adding, removing, and reordering field definitions. Everything runs locally — no external services, accounts, or network APIs are involved. ## Requirements Implement a Reflex app served at the index route `/` with two side-by-side panels: **Builder panel** (left): * A text input to type a new field's label, a selector to pick the new field's type (one of exactly `text`, `number`, `select`), and a button to append the field to the form definition. * The current list of field definitions, rendered in order. import reflex as rx
from pydantic import BaseModel

class FieldDef(BaseModel):
    label: str = ''
    type: str = 'text'
    required: bool = False

class FormState(rx.State):
    fields: list[FieldDef] = []
    form_values: dict[str, str] = {}
    new_label: str = ''
    new_type: str = 'text'
    error_message: str = ''
    success_message: str = ''

    def set_field_value(self, label_value: list):
        if isinstance(label_value, list) and len(label_value) == 2:
            label, value = label_value[0], label_value[1]
            self.form_values = {**self.form_values, label: value}

def preview_field(field: FieldDef) -> rx.Component:
    return rx.box(
        rx.match(
            field.type,
            (
                'text',
                rx.input(
                    placeholder=field.label,
                    value=FormState.form_values[field.label],
                    on_change=lambda val: FormState.set_field_value([field.label, val]),
                ),
            ),
            rx.input(placeholder=field.label),
        ),
    )

def index() -> rx.Component:
    return rx.box(
        rx.heading('Dynamic Form Builder'),
        rx.foreach(FormState.fields, preview_field),
    )
